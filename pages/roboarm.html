   5-DOF + Gripper · Arduino UNO · 115200
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━ */

constexpr uint8_t PIN_BASE=9, PIN_SH=10, PIN_EL=11;
constexpr uint8_t PIN_WP=12, PIN_WR=13, PIN_GR=3;
constexpr float L1=120.f, L2=100.f; // link mm
constexpr uint16_t DT=20;          // loop ms

Servo sBase,sSh,sEl,sWp,sWr,sGr;
struct Pose{float b=90,s=90,e=90,wp=90,wr=90,g=90;};
Pose cur, tgt;
float spd=1.5f;

void setup(){
  Serial.begin(115200);
  sBase.attach(PIN_BASE,500,2500); sSh.attach(PIN_SH,500,2500);
  sEl.attach(PIN_EL,500,2500);   sWp.attach(PIN_WP,500,2500);
  sWr.attach(PIN_WR,500,2500);   sGr.attach(PIN_GR,500,2000);
  Serial.println(F("ROBOARM_READY"));
}

void loop(){
  if(Serial.available()) parseCmd();
  stepJoints(); delay(DT);
}

inline float step(float c,float t,float s){
  float d=t-c;
  return fabsf(d)<s ? t : c+(d>0?s:-s);
}
void stepJoints(){
  cur.b=step(cur.b,tgt.b,spd); cur.s=step(cur.s,tgt.s,spd);
  cur.e=step(cur.e,tgt.e,spd); cur.wp=step(cur.wp,tgt.wp,spd);
  cur.wr=step(cur.wr,tgt.wr,spd); cur.g=step(cur.g,tgt.g,spd);
  sBase.write(cur.b); sSh.write(cur.s); sEl.write(cur.e);
  sWp.write(cur.wp);  sWr.write(cur.wr); sGr.write(cur.g);
}
void parseCmd(){
  String ln=Serial.readStringUntil('\n'); ln.trim();
  if(ln=="HOME"){tgt={90,90,90,90,90,90};return;}
  auto get=[&](const char*k)->float{
    int i=ln.indexOf(k);
    return i<0?-1:constrain(ln.substring(i+strlen(k)).toFloat(),0,180);
  };
  if(get("B:")>=0)tgt.b=get("B:");  if(get("S:")>=0)tgt.s=get("S:");
  if(get("E:")>=0)tgt.e=get("E:");  if(get("WP:")>=0)tgt.wp=get("WP:");
  if(get("WR:")>=0)tgt.wr=get("WR:");if(get("G:")>=0)tgt.g=get("G:");
}
// ik_solver.hpp — Geometric IK, C++17
#pragma once
#include <cmath>
#include <algorithm> // std::clamp

namespace RoboArm {
inline constexpr double D2R=3.14159265/180.0, R2D=180.0/3.14159265;

struct Vec3{double x{},y{},z{};};
struct IKResult{
  double base_deg{},shoulder_deg{},elbow_deg{};
  bool   reachable{};
  double reach_ratio{};
};

class Solver{
public:
  const double L1,L2;
  constexpr Solver(double l1,double l2):L1{l1},L2{l2}{}

  [[nodiscard]] IKResult solve(Vec3 p,bool elbowUp=true) const noexcept {
    IKResult r{};
    r.base_deg=std::atan2(p.x,p.z)*R2D; // base rotation
    double h=std::sqrt(p.x*p.x+p.z*p.z);
    double D=std::sqrt(h*h+p.y*p.y);
    r.reach_ratio=D/(L1+L2);
    if(D>L1+L2+0.01||D<std::abs(L1-L2)){return r;} // unreachable
    double c2=std::clamp((D*D-L1*L1-L2*L2)/(2.0*L1*L2),-1.0,1.0);
    double t2=elbowUp?std::acos(c2):-std::acos(c2);
    double k1=L1+L2*std::cos(t2), k2=L2*std::sin(t2);
    double t1=std::atan2(p.y,h)-std::atan2(k2,k1);
    r.shoulder_deg=t1*R2D; r.elbow_deg=t2*R2D; r.reachable=true;
    return r;
  }

  [[nodiscard]] Vec3 fk(double bD,double sD,double eD) const noexcept {
    double b=bD*D2R,s=sD*D2R,e=eD*D2R;
    double r=L1*std::cos(s)+L2*std::cos(s+e);
    double y=L1*std::sin(s)+L2*std::sin(s+e);
    return {r*std::sin(b),y,r*std::cos(b)};
  }
};
} // namespace RoboArm
// pc_controller.cpp — Serial bridge, C++20
#include "ik_solver.hpp"
#include "serial_port.h"
#include <format>
#include <thread>
#include <chrono>
using namespace RoboArm;
using namespace std::chrono_literals;

/* Protocol: "B:{f},S:{f},E:{f},WP:{f},WR:{f},G:{f}\n" */

class ArmCtrl {
  SerialPort port_; Solver ik_{120,100};
public:
  explicit ArmCtrl(std::string_view dev):port_{dev,115200}{}

  void send(double b,double s,double e,double wp,double wr,double g=90){
    port_.writeLine(std::format(
      "B:{:.1f},S:{:.1f},E:{:.1f},WP:{:.1f},WR:{:.1f},G:{:.1f}",b,s,e,wp,wr,g));
  }

  bool moveTo(Vec3 p,double wp=90,double wr=90){
    auto r=ik_.solve(p);
    if(!r.reachable)return false;
    send(r.base_deg+90,r.shoulder_deg+90,r.elbow_deg+90,wp,wr);
    return true;
  }

  void home(){port_.writeLine("HOME");}

  // Sort colored block: pick from src, drop in bin
  void sortBlock(Vec3 src, Vec3 bin){
    moveTo({src.x,src.y+50,src.z}); std::this_thread::sleep_for(700ms);
    moveTo(src);                     std::this_thread::sleep_for(500ms);
    send(90,90,90,90,90,25);         std::this_thread::sleep_for(600ms);
    moveTo({src.x,src.y+60,src.z}); std::this_thread::sleep_for(500ms);
    moveTo({bin.x,bin.y+60,bin.z}); std::this_thread::sleep_for(700ms);
    moveTo(bin);                     std::this_thread::sleep_for(500ms);
    send(90,90,90,90,90,90);         std::this_thread::sleep_for(400ms);
    home();
  }
};
// pid.hpp — Joint PID with anti-windup
#pragma once
/* ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   PID — closed-loop servo control
   Requires feedback: encoder or pot
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━ */
class PID{
public:
  float Kp,Ki,Kd,setpoint=0;
  float outMin=-255,outMax=255,iLim=80;
private:
  float integral_=0,prevMeas_=0;
public:
  PID(float p,float i,float d):Kp{p},Ki{i},Kd{d}{}
  float update(float meas,float dt){
    float err=setpoint-meas;
    float P=Kp*err;
    integral_=constrain(integral_+err*dt,-iLim,iLim);
    float I=Ki*integral_;
    float D=Kd*(prevMeas_-meas)/dt; // derivative on measurement
    prevMeas_=meas;
    return constrain(P+I+D,outMin,outMax);
  }
  void reset(){integral_=prevMeas_=0;}
};
/* TUNING:
   1. Ki=Kd=0, raise Kp until oscillates
   2. Kp = 60% of oscillation value
   3. Add Ki slowly → kill steady-state error
   4. Add Kd to damp overshoot (watch noise!) */
