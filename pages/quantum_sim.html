# quantum_state.py — State vector representation using NumPy
# The simulator tracks a single qubit as a complex vector |ψ⟩ = α|0⟩ + β|1⟩

import numpy as np

# A qubit state is a 2-element complex NumPy array: [α, β]
# |0⟩ = [1, 0],  |1⟩ = [0, 1]

class QubitState:
    def __init__(self, alpha=1.0+0j, beta=0.0+0j):
        self.vec = np.array([alpha, beta], dtype=complex)
        self.normalize()

    def normalize(self):
        n = np.linalg.norm(self.vec)
        if n > 1e-10:
            self.vec /= n

    @property
    def prob_0(self):
        return np.abs(self.vec[0])**2   # |α|²

    @property
    def prob_1(self):
        return np.abs(self.vec[1])**2   # |β|²

    @property
    def bloch_vector(self):
        # Returns (x, y, z) coordinates on the Bloch sphere
        a, b = self.vec
        x = 2 * np.real(a * np.conj(b))
        y = 2 * np.imag(a * np.conj(b))
        z = np.abs(a)**2 - np.abs(b)**2
        return np.array([x, y, z])

    def __repr__(self):
        a, b = self.vec
        return f"({a:.3f})|0⟩ + ({b:.3f})|1⟩"

# ── Usage ──────────────────────────────────────────────────────────────────
psi = QubitState()                    # starts in |0⟩
print(psi)                             # (1.000+0j)|0⟩ + (0.000+0j)|1⟩
print("P(|0⟩):", psi.prob_0)           # 1.0
print("Bloch:", psi.bloch_vector)      # [0. 0. 1.]  → north pole

# gates.py — Quantum gate matrices as NumPy 2×2 complex arrays

import numpy as np

I2  = 1 / np.sqrt(2)

# ── Pauli Gates ─────────────────────────────────────────────────────────────
X = np.array([[0, 1],
               [1, 0]], dtype=complex)   # bit flip: |0⟩↔|1⟩

Y = np.array([[0, -1j],
               [1j, 0]], dtype=complex)  # bit + phase flip

Z = np.array([[1, 0],
               [0, -1]], dtype=complex)  # phase flip: |1⟩→-|1⟩

# ── Hadamard ─────────────────────────────────────────────────────────────────
H = I2 * np.array([[1,  1],
                    [1, -1]], dtype=complex)  # creates superposition

# ── Phase Gates ───────────────────────────────────────────────────────────────
S  = np.array([[1, 0],
                [0, 1j]], dtype=complex)    # +π/2 phase on |1⟩

Sd = np.array([[1, 0],
                [0, -1j]], dtype=complex)   # S† (S-dagger): -π/2 phase

T  = np.array([[1, 0],
                [0, np.exp(1j * np.pi / 4)]])  # π/8 gate

Td = T.conj().T                          # T† — conjugate transpose

# ── Apply gate to state vector ───────────────────────────────────────────────
def apply_gate(gate: np.ndarray, state: np.ndarray) -> np.ndarray:
    """Matrix-vector multiply and renormalize."""
    result = gate @ state
    return result / np.linalg.norm(result)

# ── Example: H then X ──────────────────────────────────────────────────────
psi   = np.array([1, 0], dtype=complex)  # |0⟩
psi   = apply_gate(H, psi)              # → |+⟩
psi   = apply_gate(X, psi)              # → |−⟩
print(np.round(psi, 4))                 # [ 0.7071 -0.7071]

# Verify gate is unitary: U†U = I
assert np.allclose(H @ H.conj().T, np.eye(2)), "H is not unitary?!"
print("All gates unitary ✓")             # (they are, calm down)

# rotation_gates.py — Rx, Ry, Rz using scipy and NumPy
# Rotation gates rotate the state vector on the Bloch sphere by angle θ

import numpy as np
from scipy.linalg import expm

# ── Pauli matrices (generators of rotations) ────────────────────────────────
σx = np.array([[0, 1], [1,  0]], dtype=complex)
σy = np.array([[0, -1j], [1j, 0]], dtype=complex)
σz = np.array([[1, 0], [0, -1]], dtype=complex)

# ── Rotation gate constructors ───────────────────────────────────────────────
# Rx(θ) = exp(-i·θ/2·σx) = cos(θ/2)I - i·sin(θ/2)·σx
def Rx(theta):
    c, s = np.cos(theta/2), np.sin(theta/2)
    return np.array([[c,    -1j*s],
                      [-1j*s, c   ]], dtype=complex)

def Ry(theta):
    c, s = np.cos(theta/2), np.sin(theta/2)
    return np.array([[c,  -s],
                      [s,   c]], dtype=complex)

def Rz(theta):
    return np.array([[np.exp(-1j*theta/2), 0            ],
                      [0,              np.exp(1j*theta/2)]])

# ── General rotation via matrix exponential ──────────────────────────────────
def Rn(theta, nx, ny, nz):
    """Rotate by θ around arbitrary unit axis n̂ = (nx, ny, nz)."""
    n_dot_sigma = nx*σx + ny*σy + nz*σz
    return expm(-1j * theta/2 * n_dot_sigma)

# ── Example: rotate |0⟩ by π/2 around Y (= Hadamard-like) ──────────────────
psi = np.array([1, 0], dtype=complex)
psi = Ry(np.pi/2) @ psi
print(np.round(psi, 4))  # [0.7071, 0.7071] → equal superposition

psi = Rz(np.pi) @ psi
print(np.round(psi, 4))  # [0.7071, -0.7071] → phase flip on |1⟩

# measurement.py — Probabilistic wave function collapse
# Measurement is irreversible. The universe doesn't offer undo.

import numpy as np

def measure(state: np.ndarray) -> tuple[int, np.ndarray]:
    """
    Perform a projective measurement in the computational basis.

    Returns:
        outcome  — 0 or 1 (random, weighted by |α|² and |β|²)
        new_state — collapsed state vector (|0⟩ or |1⟩)
    """
    p0 = np.abs(state[0])**2        # P(|0⟩) = |α|²
    p1 = np.abs(state[1])**2        # P(|1⟩) = |β|²

    # Sanity check — probabilities must sum to 1
    assert np.isclose(p0 + p1, 1.0), f"Norm broken: {p0+p1:.6f}"

    # Collapse — np.random.choice picks outcome weighted by probabilities
    outcome = np.random.choice([0, 1], p=[p0, p1])

    # Post-measurement state (projection + renormalization)
    if outcome == 0:
        new_state = np.array([1, 0], dtype=complex)  # |0⟩
    else:
        new_state = np.array([0, 1], dtype=complex)  # |1⟩

    return outcome, new_state

# ── Statistics over many shots ───────────────────────────────────────────────
def sample(state: np.ndarray, shots=1000) -> dict:
    """Simulate repeated measurement (re-prepare same state each time)."""
    p0 = np.abs(state[0])**2
    counts = np.random.binomial(shots, p0)
    return {'|0⟩': counts, '|1⟩': shots - counts}

# ── Example: measure a superposition state ──────────────────────────────────
psi = np.array([1, 1], dtype=complex) / np.sqrt(2)   # |+⟩

outcome, collapsed = measure(psi)
print(f"Measured: |{outcome}⟩")      # 0 or 1, 50/50
print(f"Collapsed to: {collapsed}")   # wave function is gone

stats = sample(psi, shots=10000)
print("Counts over 10000 shots:", stats)
# → {'|0⟩': ~5000, '|1⟩': ~5000}  (approximately — it's quantum, not a vending machine)

# bloch_sphere.py — 3D visualization of qubit state using Matplotlib
# Plots the Bloch sphere and state vector in real 3D space

import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

def state_to_bloch(state):
    """Convert complex state vector [α, β] to Bloch vector (x, y, z)."""
    a, b = state
    x = 2 * np.real(a * np.conj(b))
    y = 2 * np.imag(a * np.conj(b))
    z = np.abs(a)**2 - np.abs(b)**2
    return np.array([x, y, z])

def plot_bloch(state, title="Bloch Sphere"):
    fig = plt.figure(figsize=(7, 7), facecolor='#03030a')
    ax  = fig.add_subplot(111, projection='3d')
    ax.set_facecolor('#080810')

    # Draw wire-frame sphere
    u, v = np.mgrid[0:2*np.pi:30j, 0:np.pi:20j]
    xs = np.sin(v) * np.cos(u)
    ys = np.sin(v) * np.sin(u)
    zs = np.cos(v)
    ax.plot_wireframe(xs, ys, zs, color='#39FF1415', linewidth=0.3)

    # Draw axes
    for (dx, dy, dz), lbl in [
        ((1,0,0), 'X'), ((0,1,0), 'Y'),
        ((0,0,1), '|0⟩'), ((0,0,-1), '|1⟩')
    ]:
        ax.quiver(0,0,0, dx,dy,dz, color='#39FF1440', linewidth=1)
        ax.text(dx*1.15, dy*1.15, dz*1.15, lbl,
               color='#39FF14', fontsize=9)

    # Draw state vector
    bv = state_to_bloch(state)
    ax.quiver(0, 0, 0, bv[0], bv[1], bv[2],
              color='#39FF14', linewidth=2.5, arrow_length_ratio=0.15)
    ax.scatter(*bv, color='#39FF14', s=60, zorder=10)

    ax.set_title(title, color='#d8d8e8', fontsize=11)
    ax.set_xlim([-1.2, 1.2]); ax.set_ylim([-1.2, 1.2]); ax.set_zlim([-1.2, 1.2])
    plt.tight_layout(); plt.show()

# ── Plot superposition state ─────────────────────────────────────────────────
psi = np.array([1, 1j]) / np.sqrt(2)   # |i⟩ state
plot_bloch(psi, title="State |i⟩ = (|0⟩ + i|1⟩)/√2")
# → Point on equator, positive Y axis

# full_circuit.py — Complete quantum circuit: H → T → Rz → Measure
# Exactly what the simulator does when you click those buttons,
# but written in Python for people who like to suffer productively.

import numpy as np
import matplotlib.pyplot as plt

I2 = 1 / np.sqrt(2)

# ── Gate library ─────────────────────────────────────────────────────────────
GATES = {
    'X': np.array([[0,1],[1,0]], dtype=complex),
    'Y': np.array([[0,-1j],[1j,0]], dtype=complex),
    'Z': np.array([[1,0],[0,-1]], dtype=complex),
    'H': I2 * np.array([[1,1],[1,-1]], dtype=complex),
    'S': np.array([[1,0],[0,1j]], dtype=complex),
    'T': np.array([[1,0],[0,np.exp(1j*np.pi/4)]])
}

def Rz(theta):
    return np.array([[np.exp(-1j*theta/2), 0],
                      [0, np.exp(1j*theta/2)]])

# ── Circuit runner ────────────────────────────────────────────────────────────
def run_circuit(ops, shots=2048):
    psi = np.array([1, 0], dtype=complex)  # |0⟩
    history = [('init', psi.copy())]

    for op in ops:
        psi = op @ psi
        psi /= np.linalg.norm(psi)
        history.append((f'{op}', psi.copy()))

    # Measure (sample many shots)
    p0 = np.abs(psi[0])**2
    counts = {'0': np.random.binomial(shots, p0),
               '1': np.random.binomial(shots, 1-p0)}
    return psi, history, counts

# ── Run: H → T → Rz(π/4) ────────────────────────────────────────────────────
final, hist, counts = run_circuit([
    GATES['H'],
    GATES['T'],
    Rz(np.pi/4)
])

print(f"Final state: {np.round(final, 4)}")
print(f"P(|0⟩) = {np.abs(final[0])**2:.4f}")
print(f"Measurement counts (2048 shots): {counts}")

# ── Plot histogram ───────────────────────────────────────────────────────────
fig, ax = plt.subplots(facecolor='#03030a')
ax.set_facecolor('#080810')
ax.bar(['|0⟩', '|1⟩'], [counts['0'], counts['1']],
       color=['#39FF14', '#00e5ff'], edgecolor='none')
ax.set_title('Measurement Distribution', color='#d8d8e8')
plt.tight_layout(); plt.show()
